```java
import java.util.*;

class Solution {
    class Task implements Comparable<Task>{
        int s, l, num;
        
        Task(int s, int l, int num){
            this.s = s;
            this.l = l;
            this.num = num;
        }
        
        public int compareTo(Task t){
            if(this.l == t.l){
                if(this.s == t.s){
                    return this.num - t.num;
                }
                return this.s - t.s;
            }
            return this.l - t.l;
        }
    }
    
    public int solution(int[][] jobs) {
        
        int totalJobCount = jobs.length;
        PriorityQueue<Task> waitingQueue = new PriorityQueue<>();
        PriorityQueue<Task> taskQueue = new PriorityQueue<>((t1, t2) -> t1.s - t2.s);
        
        for(int num = 0; num < totalJobCount; num++){
            int[] job = jobs[num];
            
            taskQueue.add(new Task(job[0], job[1], num));
        }
        
        int curTime = taskQueue.peek().s;
        int totalReturnTime = 0;
        waitingQueue.add(taskQueue.poll());
        while(!waitingQueue.isEmpty()){
            Task cur = waitingQueue.poll();
            
            curTime += cur.l;
            
            totalReturnTime += (curTime - cur.s);
            
            System.out.println("startTime: "+cur.s + " l: "+ cur.l +" curTime: "+curTime);
            
            while(!taskQueue.isEmpty()){
                if(waitingQueue.isEmpty()){
                    Task next = taskQueue.poll();
                    waitingQueue.add(next);
                    if(curTime < next.s){
                        curTime = next.s;
                        break;
                    }
                }else{
                    if(taskQueue.peek().s <= curTime){
                        waitingQueue.add(taskQueue.poll());
                    }else{
                        break;
                    }
                }
            }
        }
        
        return totalReturnTime / totalJobCount;
    }
}
```

### 틀린 이유
디스크 컨트롤러 알고리즘은 "현재 시간 기준으로 들어올 수 있는 작업을 모두 대기열에 넣는다 -> 가장 짧은 것을 꺼내서 푼다"는 흐름으로 구성할 때 가장 깔끔하고 버그가 없습니다.

## 정답 풀이
```java
import java.util.*;

class Solution {
    class Task implements Comparable<Task>{
        int s, l, num;
        
        Task(int s, int l, int num){
            this.s = s;
            this.l = l;
            this.num = num; // 1. 누락된 초기화 수정 완료!
        }
        
        public int compareTo(Task t){
            if(this.l == t.l){
                if(this.s == t.s){
                    return this.num - t.num;
                }
                return this.s - t.s;
            }
            return this.l - t.l;
        }
    }
    
    public int solution(int[][] jobs) {
        int totalJobCount = jobs.length;
        PriorityQueue<Task> waitingQueue = new PriorityQueue<>();
        
        // 요청 시간이 같을 경우를 대비해 확실하게 정렬해주는 것이 좋습니다.
        PriorityQueue<Task> taskQueue = new PriorityQueue<>((t1, t2) -> {
            if (t1.s == t2.s) return t1.l - t2.l;
            return t1.s - t2.s;
        });
        
        for(int num = 0; num < totalJobCount; num++){
            taskQueue.add(new Task(jobs[num][0], jobs[num][1], num));
        }
        
        int curTime = 0;
        int totalReturnTime = 0;
        int completedCount = 0; // 처리가 완료된 작업의 개수
        
        // 2. 전체 작업이 모두 처리될 때까지 반복
        while(completedCount < totalJobCount) {
            
            // 현재 시간(curTime) 이하에 요청된 모든 작업을 대기열(waitingQueue)로 이동
            while(!taskQueue.isEmpty() && taskQueue.peek().s <= curTime){
                waitingQueue.add(taskQueue.poll());
            }
            
            // 대기열에 작업이 없다면, 가장 빨리 요청되는 다음 작업 시간으로 타임워프
            if(waitingQueue.isEmpty()){
                curTime = taskQueue.peek().s;
            } 
            // 대기열에 작업이 있다면, 소요시간이 가장 짧은 작업을 꺼내서 처리
            else {
                Task cur = waitingQueue.poll();
                curTime += cur.l; // 작업 소요 시간만큼 현재 시간 증가
                totalReturnTime += (curTime - cur.s); // 요청부터 종료까지 걸린 시간 누적
                completedCount++; // 완료된 카운트 증가
            }
        }
        
        return totalReturnTime / totalJobCount;
    }
}****
```
