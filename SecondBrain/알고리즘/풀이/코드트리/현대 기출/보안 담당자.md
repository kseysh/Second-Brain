## 틀린 이유

출입 구조를 짤 때 `?`를 확정하고자 했었다. 하지만, ?를 확정하지 않고 풀 수 있는 문제였다.

## 정답 풀이

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine();
        String s = sc.nextLine();
        
        // 길이가 홀수면 한 명은 들어오거나 나가기만 한 것이므로 절대 짝을 맞출 수 없습니다.
        if (n % 2 == 1) {
            System.out.println("No");
            return;
        }

        int minOpen = 0; // 가능한 열린 괄호 개수의 최솟값
        int maxOpen = 0; // 가능한 열린 괄호 개수의 최댓값

        for (int i = 0; i < n; i++) {
            char c = s.charAt(i);

            if (c == '(') {
                minOpen++;
                maxOpen++;
            } else if (c == ')') {
                minOpen--;
                maxOpen--;
            } else { // c == '?'
                minOpen--; // '?'를 ')'로 사용하는 경우
                maxOpen++; // '?'를 '('로 사용하는 경우
            }

            // 가능한 열린 괄호의 최대치가 0보다 작아지면 복구 불가능 (닫힌 괄호가 너무 많음)
            if (maxOpen < 0) {
                System.out.println("No");
                return;
            }

            // 열린 괄호의 최소치는 0보다 작을 수 없음
            // 음수가 되면, 이전에 ')'로 가정했던 '?' 하나를 '('로 바꾼 것으로 취급하여 0으로 보정
            if (minOpen < 0) {
                minOpen = 0;
            }
        }

        // 끝까지 탐색했을 때, 열린 괄호를 정확히 0개로 만들 수 있다면 올바른 괄호 문자열
        if (minOpen == 0) {
            System.out.println("Yes");
        } else {
            System.out.println("No");
        }
    }
}
```
