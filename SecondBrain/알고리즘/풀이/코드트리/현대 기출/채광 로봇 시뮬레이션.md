## 나의 풀이

```java
import java.util.*;

public class Main {

    static final int EMPTY = Integer.MIN_VALUE / 2;
    static final int MAX_N = 1000;
    static final int MAX_T = 5;
    static final int[] dy = {0, 1};
    static final int[] dx = {1, 0};
    static int[][][] dp = new int[MAX_T + 2][MAX_N][MAX_N];

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int t = sc.nextInt() + 1;
        int[][] grid = new int[n][n];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                grid[i][j] = sc.nextInt();
            }
        }
        
        for(int y = 0; y < n; y++)
            for(int x = 0; x < n; x++)
                Arrays.fill(dp[y][x], EMPTY);
        
        dp[0][0][0] = grid[0][0];
        Queue<int[]> q = new ArrayDeque<>();

        q.add(new int[]{0, 0, 0});

        while(!q.isEmpty()){
            int[] cur = q.poll();
            int r = cur[0];
            int y = cur[1];
            int x = cur[2];

            if(dp[r][y][x] == EMPTY) continue;
            // 시간 역행은 한 번 사용했다면, 무조건 r == t - 1일 때까지 시간 역행을 사용해야 한다.
            for(int d = 0; d < 2; d++){
                int ny = y + dy[d];
                int nx = x + dx[d];
                if(!inRange(n, ny, nx)) continue;
                
                if(r == 0){ // 시간 역행을 안 썼으면 둘 다 가능
                    dp[0][ny][nx] = Math.max(dp[0][ny][nx], dp[r][y][x] + grid[ny][nx]);
                    q.add(new int[]{0, ny, nx});
                    dp[1][ny][nx] = Math.max(dp[1][ny][nx], dp[r][y][x] + 2 * grid[ny][nx]);
                    q.add(new int[]{1, ny, nx});

                }else if(r == t){ // 시간 역행을 다 썼다면 무조건 안 역행
                    dp[r][ny][nx] = Math.max(dp[r][ny][nx], dp[r][y][x] + grid[ny][nx]);
                    q.add(new int[]{r, ny, nx});

                }else{// 시간 역행을 사용하고 있을 때는 무조건 역행 
                    dp[r + 1][ny][nx] = Math.max(dp[r + 1][ny][nx], dp[r][y][x] + 2 * grid[ny][nx]);
                    q.add(new int[]{r + 1, ny, nx});
                }
            }
        }
        
        // 시간 역행을 아예 사용하지 않았거나, 모두 연속해서 사용했거나.
        System.out.println(Math.max(dp[0][n - 1][n - 1], dp[t][n - 1][n - 1]));
    }

    static boolean inRange(int n, int y, int x){
        return y >= 0 && x >= 0 && y < n && x < n;
    }
}
```

### 틀린 이유

시간 역행 장치가 온 길로만 돌아갔을 때 최대 이득을 볼 수 있을 것이라고 착각했다. 최대한의 이득을 본 후 손해를 볼 수 있는 구간에서 시간 역행을 진행하고 다른 길로 갈 수 있는 방법이 존재하기 때문에, 틀린 접근이었다.

따라서 두 번째로 고민했던 것이 T가 어차피 5까지이므로 특정 순간에서 갈 수 있는 모든 곳을 가본 이후에 DP에 작성하는 방법이 있고, 결국 그 방법이 정답이었다. 즉, 각 칸에 대해 한 번의 시간 역행으로 얻을 수 있는 최대 이익을 미리 계산하는 것

## 정답 코드

```java
import java.util.Scanner;

public class Main {

    // 격자의 최대 크기를 정의합니다.
    static final int MAX_N = 1005;
    // 가능한 최소한의 수익을 정의합니다.
    static final int MIN_VALUE = -1000000000;

    static int n, t;                     // 격자의 크기와 시간 역행 시간을 나타낼 변수입니다.
    static int[][] a = new int[MAX_N][MAX_N]; // 각 격자 칸의 광물 가치/장애물 크기를 저장할 배열입니다.
    static int[][] mx = new int[MAX_N][MAX_N]; // 해당 위치에서 가능한 최대 이익을 저장할 배열입니다.
    static int[][][] dp = new int[MAX_N][MAX_N][2]; // 동적 계획법을 사용하여 최대 수익을 계산할 배열입니다.

    // 위치 (i, j)에서부터 시작하여 t초 동안 최대 이득을 계산하는 함수입니다.
    static void calculateMaxProfit(int i, int j, int x, int y, int passedTime, int profit) {
        if (passedTime == t) {
            mx[i][j] = Math.max(mx[i][j], profit);  // 최대 이득을 갱신합니다.
            return;
        }

        // 오른쪽으로 한 칸 이동할 수 있는 경우 재귀적으로 이동합니다.
        if (x + 1 <= n) {
            calculateMaxProfit(i, j, x + 1, y, passedTime + 1, profit + a[x + 1][y]);
        }

        // 아래쪽으로 한 칸 이동할 수 있는 경우 재귀적으로 이동합니다.
        if (y + 1 <= n) {
            calculateMaxProfit(i, j, x, y + 1, passedTime + 1, profit + a[x][y + 1]);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // 격자의 크기와 시간 역행 시간 T를 입력받습니다.
        n = sc.nextInt();
        t = sc.nextInt();
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                a[i][j] = sc.nextInt();
            }
        }

        // 초기화 과정입니다. 격자의 각 칸마다 계산된 최대 수익과 동적 계획법 배열을 초기화합니다.
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                mx[i][j] = MIN_VALUE;
                dp[i][j][0] = dp[i][j][1] = MIN_VALUE;
            }
        }

        // 각 위치 (i, j)에서 가능한 이득을 계산합니다.
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                calculateMaxProfit(i, j, i, j, 0, a[i][j]);
            }
        }

        // 시작 위치 (1, 1)의 이익을 초기화합니다.
        dp[1][1][0] = a[1][1];

        // 동적 계획법을 사용하여 최대 수익을 갱신합니다.
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n; j++) {
                // 시간 역행을 하지 않은 상태에서 최대 수익을 갱신합니다.
                dp[i][j][1] = Math.max(dp[i][j][1], dp[i][j][0] + mx[i][j]);

                // 아래로 한 칸 이동하면서 최대 수익을 갱신합니다.
                if (i + 1 <= n) {
                    dp[i + 1][j][0] = Math.max(dp[i + 1][j][0], dp[i][j][0] + a[i + 1][j]);
                    dp[i + 1][j][1] = Math.max(dp[i + 1][j][1], dp[i][j][1] + a[i + 1][j]);
                }

                // 오른쪽으로 한 칸 이동하면서 최대 수익을 갱신합니다.
                if (j + 1 <= n) {
                    dp[i][j + 1][0] = Math.max(dp[i][j + 1][0], dp[i][j][0] + a[i][j + 1]);
                    dp[i][j + 1][1] = Math.max(dp[i][j + 1][1], dp[i][j][1] + a[i][j + 1]);
                }
            }
        }

        // 결과를 출력합니다. 시간 역행을 사용한 경우와 사용하지 않은 경우 중 더 큰 수익을 출력합니다.
        System.out.println(Math.max(dp[n][n][0], dp[n][n][1]));

        sc.close();
    }
}
```
