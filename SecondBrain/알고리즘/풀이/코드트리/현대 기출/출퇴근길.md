## 나의 풀이 (틀림)

```java
import java.util.*;

public class Main {
    
    static final int NOT_REACH = 0;
    static final int REACH = 1;
    static List<Integer>[] graph;
    static List<Boolean>[] isUsed;
    static boolean[] isVisit;
    static boolean[][] canReachTarget;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); // node
        int m = sc.nextInt(); // edge
        
        graph = new ArrayList[n];
        isUsed = new ArrayList[n];
        for(int i = 0; i < n; i++){
            graph[i] = new ArrayList<>();
            isUsed[i] = new ArrayList<>();
        } 

        for (int i = 0; i < m; i++) {
            int x = sc.nextInt() - 1;
            int y = sc.nextInt() - 1;
            graph[x].add(y);
            isUsed[x].add(false);
        }
        int s = sc.nextInt() - 1;
        int e = sc.nextInt() - 1;
        
        canReachTarget = new boolean[2][n];

        for(int i = 0; i < n; i++) Collections.fill(isUsed[i], false);
        visit(s, e, 0);

        // e -> s
        for(int i = 0; i < n; i++) Collections.fill(isUsed[i], false);
        visit(e, s, 1);

        int answer = 0;
        for(int i = 0; i < n; i++){
            if(canReachTarget[0][i] && canReachTarget[1][i]) {
                answer++;
                // System.out.println(i + 1);
            }
        }

        System.out.println(answer);
    }

    // s -> e로 가면서 목적지 도달 시 canReachTarget의 type번에 경로 저장
    static boolean visit(int cur, int target, int type){
        if(cur == target) return true;

        boolean canReach = false;
        for(int i = 0; i < graph[cur].size(); i++){
            int next = graph[cur].get(i);
            if(isUsed[cur].get(i)) {
                if(canReachTarget[type][next]){ // 다음 목적지가 목적지를 찍었다면 그곳을 가는 경로도 목적지를 찍을 수 있다.
                    canReach = true;
                    canReachTarget[type][cur] = true;
                }
                continue;
            }
            isUsed[cur].set(i, true);

            if(visit(next, target, type)){ // 목적지에 도착
                canReach = true;
                canReachTarget[type][cur] = true;
            }
        }

        return canReach;
    }
}

```

### 틀린 이유

문제를 해결하기 위해서는 정방향 그래프에서 s→e로 가면서 방문한 곳들을 체크해두고, 역방향 그래프에서 e → s로 가면서 방문한 곳을 체크하는 방식으로 풀었어야 한다.

## 정답 풀이

```java
import java.io.*;
import java.util.*;

public class Main {
    static int n, m;
    static int s, e;
    static List<Integer>[] graph;
    static List<Integer>[] reverseGraph;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());

        n = Integer.parseInt(st.nextToken());
        m = Integer.parseInt(st.nextToken());

        graph = new ArrayList[n + 1];
        reverseGraph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) {
            graph[i] = new ArrayList<>();
            reverseGraph[i] = new ArrayList<>();
        }

        for (int i = 0; i < m; i++) {
            st = new StringTokenizer(br.readLine());
            int u = Integer.parseInt(st.nextToken());
            int v = Integer.parseInt(st.nextToken());
            graph[u].add(v);
            reverseGraph[v].add(u); // 역방향 그래프 생성
        }

        st = new StringTokenizer(br.readLine());
        s = Integer.parseInt(st.nextToken());
        e = Integer.parseInt(st.nextToken());

        // 1. 출근길: S에서 출발하여 도달 가능한 노드 (T에 도착하면 더 이상 이동 금지)
        boolean[] fromS = new boolean[n + 1];
        fromS[e] = true; // T를 미리 방문 처리하여 T 너머로 가지 못하게 막음
        bfs(s, graph, fromS);

        // 2. 출근길: T로 도달 가능한 노드 (역방향 그래프에서 T 출발)
        boolean[] toE = new boolean[n + 1];
        bfs(e, reverseGraph, toE);

        // 3. 퇴근길: T에서 출발하여 도달 가능한 노드 (S에 도착하면 더 이상 이동 금지)
        boolean[] fromE = new boolean[n + 1];
        fromE[s] = true; // S를 미리 방문 처리하여 S 너머로 가지 못하게 막음
        bfs(e, graph, fromE);

        // 4. 퇴근길: S로 도달 가능한 노드 (역방향 그래프에서 S 출발)
        boolean[] toS = new boolean[n + 1];
        bfs(s, reverseGraph, toS);

        // 결과 계산 (S와 T는 문제 조건상 도달 가능한 '중간' 동네 세기에서 제외해야 함)
        int count = 0;
        for (int i = 1; i <= n; i++) {
            if (i == s || i == e) continue;
            
            // 4가지 조건을 모두 만족하는 노드
            if (fromS[i] && toE[i] && fromE[i] && toS[i]) {
                count++;
            }
        }

        System.out.println(count);
    }

    static void bfs(int start, List<Integer>[] adj, boolean[] visited) {
        Queue<Integer> q = new LinkedList<>();
        
        // start 노드가 이미 방문 처리되어 있지 않다면 큐에 넣음
        if (!visited[start]) {
            visited[start] = true;
            q.add(start);
        }

        while (!q.isEmpty()) {
            int cur = q.poll();

            for (int next : adj[cur]) {
                if (!visited[next]) {
                    visited[next] = true;
                    q.add(next);
                }
            }
        }
    }
}
```
