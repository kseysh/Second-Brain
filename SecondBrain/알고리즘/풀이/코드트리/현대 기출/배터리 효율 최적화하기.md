## 나의 풀이

```java
import java.util.*;

public class Main {

    static final int MAX_N = 10;
    static final int MAX_M = 10;

    static int[] dy = {0, 1, 0, -1};
    static int[] dx = {1, 0, -1, 0};

    static boolean[][] battery1 = new boolean[MAX_N][MAX_M];
    static boolean[][] battery2 = new boolean[MAX_N][MAX_M];

    static Set<String> battery1Hash = new HashSet<>();
    static Set<String> battery2Hash = new HashSet<>();

    static int n, m;
    static int answer = Integer.MIN_VALUE;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        int[][] board = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                board[i][j] = sc.nextInt();
            }
        }
        
        backTracking(board, 0);
        System.out.println(answer);
    }

    public static void backTracking(int[][] board, int batteryCount){
        if(batteryCount == 10){ // 배터리 다 만들었으니 최댓값 확인
            caculateScore(board);
            return;
        }

        if(batteryCount < 5){
            createBatteryCell(board, battery1, batteryCount);
        }else{
            if(batteryCount == 5){
                battery2Hash.clear(); 
            }
            createBatteryCell(board, battery2, batteryCount);
        }
    }

    public static boolean checkDuplicated(boolean[][] battery, Set<String> batteryHash){
        StringBuilder sb = new StringBuilder();

        for(int y = 0; y < n; y++){
            for(int x = 0; x < m; x++){
                if(battery[y][x]){
                    sb.append(y).append(",").append(x).append("/");
                }
            }
        }

        String hashStr = sb.toString();

        if(batteryHash.contains(hashStr)) return true;
        
        batteryHash.add(hashStr);
        return false;
    }

    public static void createBatteryCell(int[][] board, boolean[][] battery, int batteryCount){
    
        for(int y = 0; y < n; y++){
            for(int x = 0; x < m; x++){
                if(battery[y][x]) continue; 

                if(batteryCount % 5 == 0){ // 처음 시작하는 배터리 셀은 아무 곳에나 만든다.
                    battery[y][x] = true;
                    backTracking(board, batteryCount + 1);
                    battery[y][x] = false;
                }else{
                    boolean isNearByBattery = false;
                    for(int d = 0; d < 4; d++){
                        int ny = y + dy[d];
                        int nx = x + dx[d];
                        if(!inRange(ny, nx)) continue;
                        if(battery[ny][nx]) isNearByBattery = true;
                    }

                    if(isNearByBattery){  // 주변에 배터리 셀이 존재할 때만 만든다.
                        battery[y][x] = true;
                        if(batteryCount < 5){
                            if(!checkDuplicated(battery, battery1Hash)){
                                backTracking(board, batteryCount + 1);
                            }
                        }else{
                            if(!checkDuplicated(battery, battery2Hash)){
                                backTracking(board, batteryCount + 1);
                            }
                        }
                        battery[y][x] = false;
                    }
                }
            }
        }
    }

    public static void caculateScore(int[][] board){
        int score = 0;
        int overlapCount = 0;

        for(int y = 0; y < n; y++){
            for(int x = 0; x < m; x++){
                if(battery1[y][x]) score += board[y][x];
                if(battery2[y][x]) score += board[y][x];
                if(battery1[y][x] && battery2[y][x]) overlapCount++;
            }
        }

        if(overlapCount == 2){
            answer = Math.max(answer, score);
        }    
    }

    public static boolean inRange(int y, int x){
        return y >= 0 && x >= 0 && y < n && x < m;
    } 
}
```

### 틀린 이유

독립적인 5칸짜리 모양을 먼저 다 구해놓고 비교했다면, 시간 초과가 발생하지 않았을 것이다.

나 또한, 수정하기 위해 HashSet을 활용해보았지만, 나의 경우 A→B→C를 set에 넣었다면 C→B→A도 막아야 하지만, C→B때까지는 막지 못하고 C→B→A에 가서야 막을 수 있는 구조이므로 불필요한 행동을 하는 것은 맞다. 또한, String이기 때문에 부하도 존재한다. ⇒ bit 연산으로 하면 더 빠를 가능성이 있다.

`contains`로 확인하고 `return`을 해서 그 순간 다음 재귀를 안 들어가는 것은 맞지만, **그 지점까지 들어가기 위해 실행된 수많은 이전 단계의 재귀 호출(과 문자열 생성 연산)들이 이미 시간을 전부 잡아먹은 상태**가 되기 때문에 시간 초과가 나게 됩니다.

### 정답 풀이

```java
import java.util.*;

public class Main {
    // 이동할 수 있는 네 방향을 정의합니다.
    private static final int[] dx = {-1, 0, 1, 0}, dy = {0, -1, 0, 1};

    private static int n, m; // 격자의 세로, 가로 길이를 저장합니다.
    private static int[][] board = new int[15][15]; // 격자의 각 셀에 저장된 에너지 출력 값을 저장합니다.
    private static int ans = Integer.MIN_VALUE; // 가능한 최대 효율 점수를 저장합니다. 초기값은 매우 낮은 값으로 설정합니다.

    // (x, y)가 격자 범위를 벗어나는지 확인하는 함수입니다.
    private static boolean isOutrange(int x, int y) {
        return !(1 <= x && x <= n && 1 <= y && y <= m);
    }

    // (x, y) 위치를 일련번호로 변환하는 함수입니다.
    private static int conv(int x, int y) {
        return (x - 1) * m + y;
    }

    // 일련번호를 (x, y) 좌표로 되돌리는 함수입니다.
    private static int[] reconv(int num) {
        int x = 1 + (num - 1) / m;
        int y = 1 + (num - 1) % m;
        return new int[]{x, y};
    }

    private static boolean[] vis = new boolean[105]; // 방문 여부를 체크하는 배열입니다.
    private static List<Integer> lef = new ArrayList<>(); // 남은 연결 가능한 셀의 일련번호를 저장하는 벡터입니다.
    private static Set<List<Integer>> se = new HashSet<>(); // 중복된 모듈 형태를 제외하기 위한 집합입니다.
    private static List<List<Integer>> ve = new ArrayList<>(); // 가능한 모든 모듈 형태를 저장하는 벡터입니다.

    private static List<Integer> v = new ArrayList<>(); // 현재 모듈 형태를 저장하는 일시적인 벡터입니다.

    // 가능한 모든 모듈 형태를 생성하는 함수입니다.
    private static void doit(int num) {
        if (num == 5) {
            List<Integer> newv = new ArrayList<>(v);
            Collections.sort(newv);
            if (!se.contains(newv)) {
                ve.add(newv);
                se.add(newv);
            }
            return;
        }

        for (int i = 0; i < lef.size(); i++) {
            int next_val = lef.get(i);
            if (vis[next_val]) continue;
            vis[next_val] = true;
            v.add(next_val);

            int[] cur = reconv(next_val);
            int x = cur[0];
            int y = cur[1];

            for (int dir = 0; dir < 4; dir++) {
                int nx = x + dx[dir];
                int ny = y + dy[dir];

                if (isOutrange(nx, ny)) continue;
                lef.add(conv(nx, ny));
            }

            doit(num + 1);

            // 백트래킹을 사용하여 이전 상태로 되돌립니다.
            for (int dir = 0; dir < 4; dir++) {
                int nx = x + dx[dir];
                int ny = y + dy[dir];

                if (isOutrange(nx, ny)) continue;
                lef.remove(lef.size() - 1);
            }
            vis[next_val] = false;
            v.remove(v.size() - 1);
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++) {
                board[i][j] = sc.nextInt();
            }

        // 가능한 모든 모듈 형태를 생성합니다.
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                int num = conv(i, j);
                vis[num] = true;
                v.add(num);

                lef.clear();
                for (int dir = 0; dir < 4; dir++) {
                    int nx = i + dx[dir];
                    int ny = j + dy[dir];

                    if (isOutrange(nx, ny)) continue;
                    lef.add(conv(nx, ny));
                }

                doit(1);

                v.remove(v.size() - 1);
                vis[num] = false;
            }
        }

        // 가능한 모든 모듈 쌍에 대하여 최대 효율 점수를 계산합니다.
        for (int i = 0; i < ve.size(); i++) {
            for (int j = i + 1; j < ve.size(); j++) {
                int dup = 0; // 두 모듈이 겹치는 셀의 수를 저장하는 변수입니다.
                for (int ii = 0; ii < 5; ii++) {
                    for (int jj = 0; jj < 5; jj++) {
                        if (ve.get(i).get(ii).equals(ve.get(j).get(jj))) {
                            dup++;
                        }
                    }
                }

                if (dup != 2) continue; // 정확히 두 셀이 겹쳐야만 합니다.
                int val = 0; // 효율 점수를 계산하는 변수입니다.
                for (int ii = 0; ii < 5; ii++) {
                    int num = ve.get(i).get(ii);
                    int[] cur = reconv(num);

                    int x = cur[0];
                    int y = cur[1];

                    val += board[x][y];
                }
                for (int jj = 0; jj < 5; jj++) {
                    int num = ve.get(j).get(jj);
                    int[] cur = reconv(num);

                    int x = cur[0];
                    int y = cur[1];

                    val += board[x][y];
                }

                ans = Math.max(ans, val); // 최대 효율 점수를 갱신합니다.
            }
        }

        System.out.println(ans); // 최대 효율 점수를 출력합니다.
    }
}
```

```java
import java.util.*;

public class Main {
    static int n, m;
    static int[][] board;
    static int[] dy = {0, 1, 0, -1};
    static int[] dx = {1, 0, -1, 0};
    
    // 유효하게 만들어진 모든 5칸짜리 배터리의 비트마스크를 저장할 셋
    static Set<Long> uniqueBatteries = new HashSet<>();
    static List<Long> batteryList;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        m = sc.nextInt();
        board = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                board[i][j] = sc.nextInt();
            }
        }

        // 1. 모든 시작점에 대해 5칸짜리 배터리 모양을 생성
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                List<Integer> selected = new ArrayList<>();
                selected.add(i * m + j);
                long initialMask = 1L << (i * m + j);
                generateBattery(selected, initialMask);
            }
        }

        // 중복이 제거된 배터리들을 리스트로 변환 (인덱스 접근용)
        batteryList = new ArrayList<>(uniqueBatteries);
        int answer = Integer.MIN_VALUE;

        // 2. 두 배 text 배터리를 조합(2개 선택)하여 조건 확인 및 점수 계산
        int size = batteryList.size();
        for (int i = 0; i < size; i++) {
            for (int j = i; j < size; j++) { // 자기 자신과 같은 모양이어도 위치가 다르면 다른 마스크를 가짐
                long b1 = batteryList.get(i);
                long b2 = batteryList.get(j);

                // 두 비트마스크를 AND 연산하여 1인 비트의 개수(겹치는 칸 수)를 센다
                long overlap = b1 & b2;
                if (Long.bitCount(overlap) == 2) {
                    int score = calculateScoreForMask(b1) + calculateScoreForMask(b2);
                    answer = Math.max(answer, score);
                }
            }
        }

        System.out.println(answer == Integer.MIN_VALUE ? -1 : answer);
    }

    // 인접한 칸들을 확장해가며 5칸짜리 조합을 만드는 함수
    public static void generateBattery(List<Integer> selected, long mask) {
        if (selected.size() == 5) {
            uniqueBatteries.add(mask);
            return;
        }

        // 현재까지 선택된 칸들과 인접한 모든 칸을 후보군으로 둡니다.
        // 순서 중복을 막기 위해, 새로 추가할 칸은 현재 탐색 중인 조합의 '가장 큰 인덱스'보다 큰 인덱스만 고려하는 기법을 씁니다.
        Set<Integer> candidates = new HashSet<>();
        for (int cell : selected) {
            int y = cell / m;
            int x = cell % m;
            for (int d = 0; d < 4; d++) {
                int ny = y + dy[d];
                int nx = x + dx[d];
                if (ny >= 0 && nx >= 0 && ny < n && nx < m) {
                    int nextCell = ny * m + nx;
                    // 아직 선택되지 않았고, 현재 조합의 어떤 원소보다 뒤에 있는 인덱스 위주로 고려 (조합 최적화)
                    if ((mask & (1L << nextCell)) == 0) {
                        candidates.add(nextCell);
                    }
                }
            }
        }

        for (int next : candidates) {
            // 중복 방지를 위한 강제 순서 부여 (선택 리스트의 마지막 값보다 큰 것만)
            if (next > selected.get(selected.size() - 1)) {
                selected.add(next);
                generateBattery(selected, mask | (1L << next));
                selected.remove(selected.size() - 1);
            }
        }
    }

    // 비트마스크에 포함된 칸들의 점수 총합을 구하는 함수
    public static int calculateScoreForMask(long mask) {
        int sum = 0;
        for (int i = 0; i < n * m; i++) {
            if ((mask & (1L << i)) != 0) {
                int y = i / m;
                int x = i % m;
                sum += board[y][x];
            }
        }
        return sum;
    }
}
```
